country_name_construction = {
	type = country
	text = {
		localization_key = country_name_construction_name
		trigger = {
			always = no
		}
	}
	text = {
		# Prefix-driven formables: use rank prefix + base name (e.g. Greater Aquitania).
		localization_key = country_name_construction_prefix_name
		trigger = {
			OR = {
				tag = AQ2
				tag = BTT
				tag = GAC
				tag = GMA
				tag = HBR
				tag = LES
				tag = LO2
				tag = LSM
				tag = MGG
				tag = PH2
				tag = UCI
			}
		}
	}
	text = {
		# Non-prefix complex names stay literal.
		localization_key = country_name_construction_name
		trigger = {
			OR = {
				tag = DL2
				tag = HPP
				tag = HSD
			}
		}
	}
	text = {
		localization_key = country_name_construction_prefix_rank_of_name_but_map_is_just_prefix_rank
		trigger = {
			always = no
		}
	}
	text = {
		localization_key = country_name_construction_prefix_name_rank
		trigger = {
			always = no
		}
	}
	text = {
		localization_key = country_name_construction_prefix_name
		trigger = {
			country_rank ?= country_rank:rank_empire
			court_language ?= {
				language_family ?= language_family:chinese_language_family
			}
		}
	}
	text = {
		localization_key = country_name_construction_prefix_name_horde
		trigger = {
			government_type = government_type:steppe_horde
		}
	}
	text = {
		localization_key = country_name_construction_prefix_adjective_rank
		trigger = {
			OR = {
				country_rank = country_rank:rank_empire
				country_type = pop
				AND = {
					government_type = government_type:monarchy
					OR = {
						# Base-derived and I:R formables with explicit monarchy requirements.
						tag = ODR
						tag = SEL
						tag = BPK
						tag = PRY
						tag = EGY
						tag = ALX
					}
				}
				AND = {
					government_type = government_type:republic
					OR = {
						tag = DEL
						tag = ACL
						tag = DLA
						tag = HLA
					}
				}
				AND = {
					government_type = government_type:tribe
					OR = {
						tag = ODR
					}
				}
			}
		}
	}
	text = {
		localization_key = country_name_construction_crown
		trigger = {
			always = no
		}
	}
	text = {
		localization_key = country_name_construction_prefix_rank_of_name
		fallback = yes
	}
}

country_name_construction_map = {
	parent = country_name_construction
	suffix = "_map"
}
