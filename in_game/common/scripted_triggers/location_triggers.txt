location_in_any_relation_to_target_country = {
	OR = {
		owner = $target$	#Owned by target
		owner = {
			OR = {
				is_subject_of = $target$
				#Diplomacy check with the target
				AND = {
					has_ruler = yes
					$target$ = { has_ruler = yes }
					ruler ?= $target$.ruler
				}
			}
		}
		#Same IO and the target is also leader of that IO
		AND = {
			is_owned_by_any_international_organization = yes
			any_international_organization_owner = {
				international_organization_has_leader = yes
				leader_country ?= $target$
			}
		}
	}
}

	OR = {
		has_owner = no
		owner ?= {
			OR = {
				NOT = { is_member_of_international_organization = international_organization:japanese_shogunate }
				is_leader_of_international_organization = international_organization:japanese_shogunate
				AND = {
					exists = scope:loser
					this = scope:loser
				}
			}
		}
	}
	scope:target_clan = {
		owns_most_foreign_buildings_in_location = prev
	}
}
