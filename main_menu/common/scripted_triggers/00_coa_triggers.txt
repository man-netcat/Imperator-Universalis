has_barbarian_flag_trigger = {
	scope:actor ?= {
		exists = culture
		culture = {
			OR = {
				has_culture_group = culture_group:ir_gallic_g
				has_culture_group = culture_group:ir_belgae_group_g
				has_culture_group = culture_group:ir_celto_pannonian_group_g
				has_culture_group = culture_group:ir_britannic_g
				has_culture_group = culture_group:ir_gaelic_g
				has_culture_group = culture_group:ir_celt_iberia_g
				has_culture_group = culture_group:ir_iberia_g
				has_culture_group = culture_group:ir_germanic_g
				has_culture_group = culture_group:ir_scythia_g
				has_culture_group = culture_group:ir_baltic_g
				has_culture_group = culture_group:ir_dacia_group_g
				has_culture_group = culture_group:ir_illyrian_group_g
				has_culture_group = culture_group:ir_proto_european_g
			}
		}
	}
}
